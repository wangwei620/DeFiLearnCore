### 智能合约结构
```solidity
// contracts/StakingPool.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import "@openzeppelin/contracts/security/ReentrancyGuard.sol";
import "@openzeppelin/contracts/access/Ownable.sol";

contract StakingPool is ReentrancyGuard, Ownable {
    IERC20 public stakingToken;
    IERC20 public rewardToken;
    
    // 年化收益率：10% = 1000（基点）
    uint public rewardRate = 1000;
    uint public constant RATE_DENOMINATOR = 10000;
    
    struct Stake {
        uint amount;
        uint timestamp;
        uint accumulatedReward;
    }
    
    mapping(address => Stake) public stakes;
    uint public totalStaked;
    
    event Staked(address indexed user, uint amount);
    event Withdrawn(address indexed user, uint amount, uint reward);
    event RewardClaimed(address indexed user, uint reward);
    
    constructor(address _stakingToken, address _rewardToken) Ownable(msg.sender) {
        stakingToken = IERC20(_stakingToken);
        rewardToken = IERC20(_rewardToken);
    }
    
    // 质押代币
    function stake(uint amount) external nonReentrant {
        require(amount > 0, "Cannot stake 0");
        
        // 先结算旧的奖励
        if (stakes[msg.sender].amount > 0) {
            _updateReward(msg.sender);
        }
        
        stakes[msg.sender].amount += amount;
        stakes[msg.sender].timestamp = block.timestamp;
        totalStaked += amount;
        
        stakingToken.transferFrom(msg.sender, address(this), amount);
        
        emit Staked(msg.sender, amount);
    }
    
    // 计算待领取奖励
    function pendingReward(address user) public view returns (uint) {
        Stake memory userStake = stakes[user];
        if (userStake.amount == 0) return 0;
        
        uint stakingDuration = block.timestamp - userStake.timestamp;
        uint reward = (userStake.amount * rewardRate * stakingDuration) 
                     / (365 days * RATE_DENOMINATOR);
        
        return userStake.accumulatedReward + reward;
    }
    
    // 内部：更新奖励
    function _updateReward(address user) internal {
        stakes[user].accumulatedReward = pendingReward(user);
        stakes[user].timestamp = block.timestamp;
    }
    
    // 领取奖励（不取出本金）
    function claimReward() external nonReentrant {
        _updateReward(msg.sender);
        
        uint reward = stakes[msg.sender].accumulatedReward;
        require(reward > 0, "No reward");
        
        stakes[msg.sender].accumulatedReward = 0;
        
        rewardToken.transfer(msg.sender, reward);
        
        emit RewardClaimed(msg.sender, reward);
    }
    
    // 取出本金和奖励
    function withdraw(uint amount) external nonReentrant {
        require(amount > 0 && stakes[msg.sender].amount >= amount, "Invalid amount");
        
        _updateReward(msg.sender);
        
        uint reward = stakes[msg.sender].accumulatedReward;
        
        stakes[msg.sender].amount -= amount;
        stakes[msg.sender].accumulatedReward = 0;
        totalStaked -= amount;
        
        stakingToken.transfer(msg.sender, amount);
        if (reward > 0) {
            rewardToken.transfer(msg.sender, reward);
        }
        
        emit Withdrawn(msg.sender, amount, reward);
    }
    
    // Owner可调整利率
    function setRewardRate(uint newRate) external onlyOwner {
        require(newRate <= 5000, "Rate too high"); // 最高50%
        rewardRate = newRate;
    }
}
```

// test/StakingPool.test.js
const { expect } = require("chai");
const { ethers } = require("hardhat");
const { time } = require("@nomicfoundation/hardhat-network-helpers");

describe("StakingPool", function () {
  let stakingToken, rewardToken, pool;
  let owner, user1, user2;
  
  beforeEach(async function () {
    [owner, user1, user2] = await ethers.getSigners();
    
    // 部署代币
    const Token = await ethers.getContractFactory("MyToken");
    stakingToken = await Token.deploy();
    rewardToken = await Token.deploy();
    
    // 部署质押池
    const Pool = await ethers.getContractFactory("StakingPool");
    pool = await Pool.deploy(stakingToken.target, rewardToken.target);
    
    // 给用户代币
    await stakingToken.transfer(user1.address, ethers.parseEther("1000"));
    await stakingToken.connect(user1).approve(pool.target, ethers.parseEther("1000"));
    
    // 给池子奖励代币
    await rewardToken.transfer(pool.target, ethers.parseEther("10000"));
  });
  
  it("Should stake tokens", async function () {
    await pool.connect(user1).stake(ethers.parseEther("100"));
    
    const stake = await pool.stakes(user1.address);
    expect(stake.amount).to.equal(ethers.parseEther("100"));
    expect(await pool.totalStaked()).to.equal(ethers.parseEther("100"));
  });
  
  it("Should calculate rewards correctly", async function () {
    await pool.connect(user1).stake(ethers.parseEther("100"));
    
    // 快进30天
    await time.increase(30 * 24 * 60 * 60);
    
    const reward = await pool.pendingReward(user1.address);
    
    // 预期奖励：100 * 10% / 365 * 30 ≈ 0.822 tokens
    expect(reward).to.be.closeTo(
      ethers.parseEther("0.822"),
      ethers.parseEther("0.01") // 允许小误差
    );
  });
  
  it("Should claim rewards", async function () {
    await pool.connect(user1).stake(ethers.parseEther("100"));
    await time.increase(30 * 24 * 60 * 60);
    
    const rewardBefore = await rewardToken.balanceOf(user1.address);
    await pool.connect(user1).claimReward();
    const rewardAfter = await rewardToken.balanceOf(user1.address);
    
    expect(rewardAfter - rewardBefore).to.be.gt(0);
  });
});

## 📁 项目2：简化版去中心化借贷协议（第4-5周完成）

### 项目概述
实现类似Aave的借贷协议，用户存入资产赚取利息，也可抵押借款。

### 核心功能
```
存款赚息 ← → 超额抵押借款 → 清算机制
```

### 智能合约架构
```solidity
// contracts/LendingPool.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import "@openzeppelin/contracts/security/ReentrancyGuard.sol";

contract SimpleLending is ReentrancyGuard {
    struct Market {
        uint totalDeposits;
        uint totalBorrows;
        uint depositRate;      // 存款年利率（基点）
        uint borrowRate;       // 借款年利率（基点）
        uint collateralFactor; // 抵押率：75% = 7500
    }
    
    struct UserAccount {
        uint deposited;
        uint borrowed;
        uint lastUpdateTime;
    }
    
    // 资产地址 => 市场数据
    mapping(address => Market) public markets;
    // 用户 => 资产 => 账户
    mapping(address => mapping(address => UserAccount)) public accounts;
    
    uint public constant LIQUIDATION_THRESHOLD = 8000; // 80%
    uint public constant LIQUIDATION_BONUS = 500;      // 5%
    
    event Deposited(address indexed user, address indexed token, uint amount);
    event Borrowed(address indexed user, address indexed token, uint amount);
    event Repaid(address indexed user, address indexed token, uint amount);
    event Liquidated(address indexed liquidator, address indexed borrower, uint amount);
    
    // 初始化市场
    function initMarket(
        address token,
        uint depositRate,
        uint borrowRate,
        uint collateralFactor
    ) external {
        require(markets[token].collateralFactor == 0, "Market exists");
        
        markets[token] = Market({
            totalDeposits: 0,
            totalBorrows: 0,
            depositRate: depositRate,
            borrowRate: borrowRate,
            collateralFactor: collateralFactor
        });
    }
    
    // 存款
    function deposit(address token, uint amount) external nonReentrant {
        require(markets[token].collateralFactor > 0, "Market not exists");
        
        _accrueInterest(msg.sender, token);
        
        accounts[msg.sender][token].deposited += amount;
        markets[token].totalDeposits += amount;
        
        IERC20(token).transferFrom(msg.sender, address(this), amount);
        
        emit Deposited(msg.sender, token, amount);
    }
    
    // 借款
    function borrow(address token, uint amount) external nonReentrant {
        _accrueInterest(msg.sender, token);
        
        require(_getHealthFactor(msg.sender) >= 1e18, "Insufficient collateral");
        
        accounts[msg.sender][token].borrowed += amount;
        markets[token].totalBorrows += amount;
        
        IERC20(token).transfer(msg.sender, amount);
        
        emit Borrowed(msg.sender, token, amount);
    }
    
    // 还款
    function repay(address token, uint amount) external nonReentrant {
        _accrueInterest(msg.sender, token);
        
        uint borrowed = accounts[msg.sender][token].borrowed;
        uint repayAmount = amount > borrowed ? borrowed : amount;
        
        accounts[msg.sender][token].borrowed -= repayAmount;
        markets[token].totalBorrows -= repayAmount;
        
        IERC20(token).transferFrom(msg.sender, address(this), repayAmount);
        
        emit Repaid(msg.sender, token, repayAmount);
    }
    
    // 清算
    function liquidate(
        address borrower,
        address collateralToken,
        address debtToken,
        uint debtAmount
    ) external nonReentrant {
        require(_getHealthFactor(borrower) < 1e18, "Cannot liquidate");
        
        _accrueInterest(borrower, debtToken);
        _accrueInterest(borrower, collateralToken);
        
        // 计算可获得的抵押品（含奖励）
        uint collateralAmount = (debtAmount * (10000 + LIQUIDATION_BONUS)) / 10000;
        
        // 扣除借款
        accounts[borrower][debtToken].borrowed -= debtAmount;
        markets[debtToken].totalBorrows -= debtAmount;
        
        // 扣除抵押品
        accounts[borrower][collateralToken].deposited -= collateralAmount;
        markets[collateralToken].totalDeposits -= collateralAmount;
        
        // 转账
        IERC20(debtToken).transferFrom(msg.sender, address(this), debtAmount);
        IERC20(collateralToken).transfer(msg.sender, collateralAmount);
        
        emit Liquidated(msg.sender, borrower, debtAmount);
    }
    
    // 计算健康因子
    function _getHealthFactor(address user) internal view returns (uint) {
        // 简化版：单一资产
        // 实际应遍历所有资产
        // healthFactor = 抵押品价值 * 抵押率 / 借款价值
        // > 1 安全，< 1 可被清算
        
        // 这里省略价格预言机集成，假设1:1
        // 生产环境需使用Chainlink
        return 1e18; // 示例
    }
    
    // 计息
    function _accrueInterest(address user, address token) internal {
        UserAccount storage account = accounts[user][token];
        if (account.lastUpdateTime == 0) {
            account.lastUpdateTime = block.timestamp;
            return;
        }
        
        uint timeElapsed = block.timestamp - account.lastUpdateTime;
        
        // 简化利息计算（实际应该复利）
        if (account.borrowed > 0) {
            uint interest = (account.borrowed * markets[token].borrowRate * timeElapsed)
                          / (365 days * 10000);
            account.borrowed += interest;
        }
        
        account.lastUpdateTime = block.timestamp;
    }
}
```

    function getCurrentPrice(uint listingId) public view returns (uint) {
        Listing memory listing = listings[listingId];
        require(listing.listingType == ListingType.DUTCH_AUCTION);
        
        uint timeElapsed = block.timestamp - listing.startTime;
        if (timeElapsed >= listing.duration) {
            return listing.endPrice;
        }
        
        uint priceDrop = listing.price - listing.endPrice;
        uint currentDrop = (priceDrop * timeElapsed) / listing.duration;
        
        return listing.price - currentDrop;
    }

    function bid(uint listingId) external payable {
        Listing storage listing = listings[listingId];
        require(listing.listingType == ListingType.ENGLISH_AUCTION);
        require(msg.value > listing.highestBid);
        
        // 退回上一个出价者
        if (listing.highestBidder != address(0)) {
            payable(listing.highestBidder).transfer(listing.highestBid);
        }
        
        listing.highestBidder = msg.sender;
        listing.highestBid = msg.value;
    }

// 实时倒计时组件
function AuctionTimer({ endTime }) {
  const [timeLeft, setTimeLeft] = useState('');
  
  useEffect(() => {
    const interval = setInterval(() => {
      const now = Date.now();
      const diff = endTime - now;
      
      if (diff <= 0) {
        setTimeLeft('已结束');
        return;
      }
      
      const hours = Math.floor(diff / 3600000);
      const mins = Math.floor((diff % 3600000) / 60000);
      const secs = Math.floor((diff % 60000) / 1000);
      
      setTimeLeft(`${hours}h ${mins}m ${secs}s`);
    }, 1000);
    
    return () => clearInterval(interval);
  }, [endTime]);
  
  return <div className="text-red-500 font-bold">{timeLeft}</div>;
}
